### ③ 원천 락을 잡고 복사한다

```
mkdir -p "$레포/.claude/skills/k-politics/DBs/원천"
# 락을 쥔 채로 VACUUM INTO — 수집기가 다시 깨어나도 끼어들 수 없다
python3 - <<'PY'
import fcntl, sqlite3, time
from pathlib import Path
H = Path("/Users/seongjin/Coding/Harness for K-Politics/.claude/skills")
D = Path("/Users/seongjin/Coding/Skills for K-Politics/.claude/skills/k-politics/DBs/원천")
쌍 = [(H/"congress/CONGRESS.db", D/"국회.db"), (H/"law/LAW.db", D/"법령.db")]
잠금 = []
try:
    for 원본, _ in 쌍:                      # 고정 순서: 국회 → 법령
        f = Path(str(원본) + ".lock").open("a+")
        fcntl.flock(f.fileno(), fcntl.LOCK_EX)   # 블로킹 — 도는 수집이 끝날 때까지 기다린다
        잠금.append(f)
    for 원본, 대상 in 쌍:
        대상.unlink(missing_ok=True)
        c = sqlite3.connect(f"file:{원본}?mode=ro", uri=True)
        t = time.monotonic()
        c.execute("VACUUM INTO ?", (str(대상), ))
        c.close()
        print(f"{대상.name} {대상.stat().st_size/2**30:.2f}GiB {time.monotonic()-t:.0f}초")
finally:
    for f in reversed(잠금):
        fcntl.flock(f.fileno(), fcntl.LOCK_UN); f.close()
PY
```

### ④ 원본과 표별 행 수가 같은지 대조한다

```
python3 - <<'PY'
import sqlite3
from pathlib import Path
H = Path("/Users/seongjin/Coding/Harness for K-Politics/.claude/skills")
D = Path("/Users/seongjin/Coding/Skills for K-Politics/.claude/skills/k-politics/DBs/원천")
for 원본, 대상 in [(H/"congress/CONGRESS.db", D/"국회.db"), (H/"law/LAW.db", D/"법령.db")]:
    a = sqlite3.connect(f"file:{원본}?mode=ro", uri=True)
    b = sqlite3.connect(f"file:{대상}?mode=ro&immutable=1", uri=True)
    assert b.execute("PRAGMA integrity_check").fetchone()[0] == "ok", 대상
    표 = [r[0] for r in a.execute("SELECT name FROM sqlite_master WHERE type='table'")]
    어긋남 = {t: (x, y) for t in 표
              if (x := a.execute(f'SELECT COUNT(*) FROM "{t}"').fetchone()[0])
              != (y := b.execute(f'SELECT COUNT(*) FROM "{t}"').fetchone()[0])}
    print(대상.name, "표", len(표), "어긋남", 어긋남 or "없음")
    assert not 어긋남
PY
```

**판정:** 두 파일 다 `integrity_check` ok, 어긋남 없음.

### ⑤ 새 경로에서 백업 한 세대를 만든다

⚠️ **Harness 세대로는 새 경로의 이행이 승인되지 않는다.** 이행 검증이 곁기록의 **원본 절대

## 이 절차를 쓰다가 겪은 것 (2026-09-12)

**`git checkout main` 이 스냅샷을 지웠다.** `법.db` 는 gitignore 지만 **초기 커밋에 0바이트로
추적돼 있었다.** 브랜치를 그 커밋 쪽으로 옮기는 순간 git 이 4.6GB 를 0바이트로 덮어썼고,
뒤이은 `pull` 이 (머지가 그 파일을 삭제했으므로) 아예 지웠다.

**한 번 추적된 적 있는 파일은 gitignore 에 넣어도 브랜치 전환에서 안전하지 않다.**
스냅샷은 3분이면 다시 만들어지니 실손은 없었지만, 원천에는 이 규칙이 그대로 적용되면 안 된다 —
`DBs/원천/` 은 **한 번도 추적된 적이 없어야 한다.** 위 ③에서 디렉터리를 그때 만드는 이유가
이것이기도 하다. C4 이후 `git ls-tree -r HEAD --name-only | grep 원천` 이 0건인지 확인해라.
